--@author Claude AI
--@shared
--@model models/hunter/plates/plate2x2.mdl

--[[
    Perfect Circle Challenge
    A multiplayer game where players compete to draw the most perfect circle!

    SETUP:
    1. Spawn this Starfall Processor chip
    2. Spawn a Starfall Screen
    3. Link the screen to this chip using the Starfall tool
    4. Players look at the screen and press USE (E) to draw!

    HOW TO PLAY:
    - Look at the screen and press E to start drawing
    - Move your mouse in a circle
    - Press E again to submit and get scored
    - Try to draw the most perfect circle!

    FEATURES:
    - Multiplayer - everyone sees all drawings
    - Real-time scoring based on circularity
    - Leaderboard showing top players
    - Visual feedback on circle quality
]]

-- Configuration
local DRAW_TIME = 4 -- seconds to keep drawings visible
local MIN_POINTS = 15 -- minimum points to draw before scoring
local CANVAS_SIZE = 512 -- size of drawing area

-- Game state (shared)
local drawings = {} -- Current active drawings
local scores = {} -- Player scores: {playerName = score}

-- Client state
if CLIENT then
    -- Per-player drawing state
    playerDrawings = {}
    playerScores = {}
end

-- Colors
local COLOR_BACKGROUND = Color(20, 20, 30)
local COLOR_GRID = Color(40, 40, 60, 100)
local COLOR_GOOD = Color(0, 255, 0)
local COLOR_OKAY = Color(255, 255, 0)
local COLOR_BAD = Color(255, 100, 0)
local COLOR_POOR = Color(255, 0, 0)

-- Helper function: Calculate distance between two points
local function distance(x1, y1, x2, y2)
    local dx = x2 - x1
    local dy = y2 - y1
    return math.sqrt(dx * dx + dy * dy)
end

-- Calculate how "perfect" a circle is (0-100 score)
local function evaluateCircle(points)
    if #points < MIN_POINTS then
        return 0, "Too few points! Draw more."
    end

    -- Find center point (average of all points)
    local centerX, centerY = 0, 0
    for i = 1, #points do
        centerX = centerX + points[i].x
        centerY = centerY + points[i].y
    end
    centerX = centerX / #points
    centerY = centerY / #points

    -- Calculate average radius
    local avgRadius = 0
    for i = 1, #points do
        avgRadius = avgRadius + distance(centerX, centerY, points[i].x, points[i].y)
    end
    avgRadius = avgRadius / #points

    -- Prevent division by zero for tiny circles
    if avgRadius < 10 then
        return 0, "Circle too small!"
    end

    -- Calculate standard deviation of radii
    local variance = 0
    for i = 1, #points do
        local radius = distance(centerX, centerY, points[i].x, points[i].y)
        local diff = radius - avgRadius
        variance = variance + (diff * diff)
    end
    variance = variance / #points
    local stdDev = math.sqrt(variance)

    -- Score based on consistency (lower stdDev = more perfect)
    local consistency = math.max(0, 100 - (stdDev / avgRadius) * 200)

    -- Check if drawing is closed (start and end points close)
    local closeness = 0
    if #points >= 2 then
        local startPoint = points[1]
        local endPoint = points[#points]
        local gap = distance(startPoint.x, startPoint.y, endPoint.x, endPoint.y)
        closeness = math.max(0, 100 - (gap / avgRadius) * 100)
    end

    -- Final score (weighted average)
    local finalScore = math.floor((consistency * 0.7 + closeness * 0.3))
    finalScore = math.max(0, math.min(100, finalScore))

    -- Generate feedback message
    local message = ""
    if finalScore >= 90 then
        message = "PERFECT! Circle master!"
    elseif finalScore >= 75 then
        message = "Great circle! Very round!"
    elseif finalScore >= 50 then
        message = "Not bad! Keep practicing."
    elseif finalScore >= 25 then
        message = "Looks more like an egg..."
    else
        message = "That's... abstract art?"
    end

    return finalScore, message
end

-- Get color based on score
local function getScoreColor(score)
    if score >= 75 then return COLOR_GOOD
    elseif score >= 50 then return COLOR_OKAY
    elseif score >= 25 then return COLOR_BAD
    else return COLOR_POOR
    end
end

--------------------------------------------------------------------------------
-- SERVER
--------------------------------------------------------------------------------
if SERVER then
    -- Network messages
    net.receive("submitDrawing", function(len, ply)
        local points = net.readTable()
        local score = net.readFloat()
        local playerName = ply:getName()
        local steamID = ply:getSteamID()

        -- Update player's best score
        if not scores[steamID] or score > scores[steamID].score then
            scores[steamID] = {
                name = playerName,
                score = score
            }
        end

        -- Broadcast to all clients
        net.start("receiveDrawing")
        net.writeString(playerName)
        net.writeTable(points)
        net.writeFloat(score)
        net.send()
    end)

    -- Send scores on request
    net.receive("requestScores", function(len, ply)
        net.start("receiveScores")
        net.writeTable(scores)
        net.send(ply)
    end)
end

--------------------------------------------------------------------------------
-- CLIENT
--------------------------------------------------------------------------------
if CLIENT then
    -- Local drawing state for this client
    local myCurrentPoints = {}
    local isDrawingNow = false
    local lastMousePos = {x = 0, y = 0}

    -- Receive drawing from server
    net.receive("receiveDrawing", function()
        local playerName = net.readString()
        local points = net.readTable()
        local score = net.readFloat()

        -- Add to drawings list
        table.insert(playerDrawings, {
            playerName = playerName,
            points = points,
            score = score,
            time = timer.curtime(),
            color = getScoreColor(score)
        })
    end)

    -- Receive scores from server
    net.receive("receiveScores", function()
        playerScores = net.readTable()
    end)

    -- Request scores on start
    hook.add("initialize", "requestScores", function()
        timer.simple(0.5, function()
            net.start("requestScores")
            net.send()
        end)
    end)

    -- Handle key press to toggle drawing
    hook.add("KeyPress", "toggleDrawing", function(ply, key)
        if ply ~= player() then return end

        if key == KEY.E then
            if not isDrawingNow then
                -- Start drawing
                isDrawingNow = true
                myCurrentPoints = {}
                print("Started drawing! Move your mouse in a circle, press E to submit!")
            else
                -- End drawing and submit
                isDrawingNow = false

                if #myCurrentPoints >= MIN_POINTS then
                    local score, message = evaluateCircle(myCurrentPoints)

                    -- Send to server
                    net.start("submitDrawing")
                    net.writeTable(myCurrentPoints)
                    net.writeFloat(score)
                    net.send()

                    print(message .. " Score: " .. score)

                    -- Add to local display
                    table.insert(playerDrawings, {
                        playerName = player():getName(),
                        points = myCurrentPoints,
                        score = score,
                        time = timer.curtime(),
                        color = getScoreColor(score)
                    })
                else
                    print("Not enough points! Need at least " .. MIN_POINTS)
                end

                myCurrentPoints = {}
            end
        end
    end)

    -- Track mouse movement while drawing
    hook.add("think", "trackDrawing", function()
        if isDrawingNow then
            local x, y = render.cursorPos()

            if x and y then
                -- Clamp to canvas
                x = math.clamp(x, 0, CANVAS_SIZE)
                y = math.clamp(y, 0, CANVAS_SIZE)

                -- Only add if moved enough
                if #myCurrentPoints == 0 then
                    table.insert(myCurrentPoints, {x = x, y = y})
                    lastMousePos = {x = x, y = y}
                else
                    local dist = distance(lastMousePos.x, lastMousePos.y, x, y)
                    if dist > 2 then
                        table.insert(myCurrentPoints, {x = x, y = y})
                        lastMousePos = {x = x, y = y}
                    end
                end
            end
        end
    end)

    -- Main render loop
    hook.add("render", "drawGame", function()
        -- Clear background
        render.setColor(COLOR_BACKGROUND)
        render.drawRect(0, 0, CANVAS_SIZE, CANVAS_SIZE)

        -- Draw grid
        render.setColor(COLOR_GRID)
        for i = 0, CANVAS_SIZE, 64 do
            render.drawLine(i, 0, i, CANVAS_SIZE)
            render.drawLine(0, i, CANVAS_SIZE, i)
        end

        -- Draw center guide circle
        render.setColor(Color(100, 100, 150, 30))
        render.drawCircle(CANVAS_SIZE/2, CANVAS_SIZE/2, 150)

        -- Title and instructions
        render.setColor(Color(255, 255, 255))
        render.setFont("DermaLarge")
        render.drawText(CANVAS_SIZE/2, 15, "PERFECT CIRCLE CHALLENGE", 1)

        render.setFont("DermaDefault")
        render.setColor(Color(200, 200, 255))
        render.drawText(CANVAS_SIZE/2, 45, "Press E to start drawing, E again to submit!", 1)

        -- Draw all active drawings
        local currentTime = timer.curtime()
        for i = #playerDrawings, 1, -1 do
            local drawing = playerDrawings[i]

            -- Remove old drawings
            if currentTime - drawing.time > DRAW_TIME then
                table.remove(playerDrawings, i)
            else
                -- Fade out
                local age = currentTime - drawing.time
                local alpha = math.floor(255 * (1 - age / DRAW_TIME))

                -- Draw lines
                local col = drawing.color
                render.setColor(Color(col.r, col.g, col.b, alpha))

                for j = 1, #drawing.points - 1 do
                    local p1 = drawing.points[j]
                    local p2 = drawing.points[j + 1]
                    render.drawLine(p1.x, p1.y, p2.x, p2.y)
                end

                -- Draw score label near the drawing
                if #drawing.points > 0 then
                    local centerX, centerY = 0, 0
                    for _, p in ipairs(drawing.points) do
                        centerX = centerX + p.x
                        centerY = centerY + p.y
                    end
                    centerX = centerX / #drawing.points
                    centerY = centerY / #drawing.points

                    render.setColor(Color(255, 255, 255, alpha))
                    render.setFont("DermaDefault")
                    render.drawText(centerX, centerY - 15, drawing.playerName, 1)
                    render.drawText(centerX, centerY + 5, math.floor(drawing.score), 1)
                end
            end
        end

        -- Draw current in-progress drawing
        if isDrawingNow and #myCurrentPoints > 0 then
            render.setColor(Color(255, 255, 255, 255))
            for i = 1, #myCurrentPoints - 1 do
                local p1 = myCurrentPoints[i]
                local p2 = myCurrentPoints[i + 1]
                render.drawLine(p1.x, p1.y, p2.x, p2.y)
            end

            -- Show point count
            render.setColor(Color(255, 255, 0))
            render.setFont("DermaDefault")
            render.drawText(10, 75, "Points: " .. #myCurrentPoints .. " / " .. MIN_POINTS, 0)

            if #myCurrentPoints >= MIN_POINTS then
                render.setColor(Color(0, 255, 0))
                render.drawText(10, 95, "Ready! Press E to submit", 0)
            end
        end

        -- Draw leaderboard
        local leaderY = CANVAS_SIZE - 210
        render.setColor(Color(0, 0, 0, 200))
        render.drawRect(10, leaderY, 230, 200)

        render.setColor(Color(255, 255, 255))
        render.setFont("DermaDefaultBold")
        render.drawText(125, leaderY + 10, "LEADERBOARD", 1)

        -- Sort scores
        local sortedScores = {}
        for steamID, data in pairs(playerScores) do
            table.insert(sortedScores, data)
        end
        table.sort(sortedScores, function(a, b) return a.score > b.score end)

        -- Display top 5
        render.setFont("DermaDefault")
        local y = leaderY + 35
        for i = 1, math.min(5, #sortedScores) do
            local entry = sortedScores[i]
            local color = getScoreColor(entry.score)
            render.setColor(color)
            render.drawText(20, y, i .. ". " .. entry.name .. ": " .. math.floor(entry.score), 0)
            y = y + 25
        end

        if #sortedScores == 0 then
            render.setColor(Color(150, 150, 150))
            render.drawText(125, y, "No scores yet!", 1)
        end
    end)
end

print("Perfect Circle Challenge loaded!")
print("Link this chip to a Starfall Screen, then look at the screen and press E to play!")
