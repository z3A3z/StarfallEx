--[[
    Perfect Circle Challenge

    A multiplayer game where players compete to draw the most perfect circle!

    HOW TO PLAY:
    - Look at the screen chip
    - Hold left mouse button and drag to draw a circle
    - Release to submit and get scored
    - Perfect circle = 100 points, imperfect = less

    FEATURES:
    - Multiplayer - everyone sees all drawings
    - Real-time scoring based on circularity
    - Leaderboard showing top players
    - Visual feedback on circle quality
]]

-- Configuration
local DRAW_TIME = 3 -- seconds to keep drawings visible
local MIN_POINTS = 10 -- minimum points to draw before scoring
local CANVAS_SIZE = 512 -- size of drawing area

-- Game state
local drawings = {} -- Current active drawings
local scores = {} -- Player scores: {playerName = score}
local currentDrawing = {} -- Current points being drawn by local player
local isDrawing = false
local startTime = 0

-- Colors
local COLOR_BACKGROUND = Color(20, 20, 30)
local COLOR_GRID = Color(40, 40, 60, 100)
local COLOR_GOOD = Color(0, 255, 0)
local COLOR_OKAY = Color(255, 255, 0)
local COLOR_BAD = Color(255, 100, 0)
local COLOR_POOR = Color(255, 0, 0)

-- Helper function: Calculate distance between two points
local function distance(x1, y1, x2, y2)
    local dx = x2 - x1
    local dy = y2 - y1
    return math.sqrt(dx * dx + dy * dy)
end

-- Calculate how "perfect" a circle is (0-100 score)
local function evaluateCircle(points)
    if #points < MIN_POINTS then
        return 0, "Too few points! Draw more."
    end

    -- Find center point (average of all points)
    local centerX, centerY = 0, 0
    for i = 1, #points do
        centerX = centerX + points[i].x
        centerY = centerY + points[i].y
    end
    centerX = centerX / #points
    centerY = centerY / #points

    -- Calculate average radius
    local avgRadius = 0
    for i = 1, #points do
        avgRadius = avgRadius + distance(centerX, centerY, points[i].x, points[i].y)
    end
    avgRadius = avgRadius / #points

    -- Prevent division by zero for tiny circles
    if avgRadius < 5 then
        return 0, "Circle too small!"
    end

    -- Calculate standard deviation of radii
    local variance = 0
    for i = 1, #points do
        local radius = distance(centerX, centerY, points[i].x, points[i].y)
        local diff = radius - avgRadius
        variance = variance + (diff * diff)
    end
    variance = variance / #points
    local stdDev = math.sqrt(variance)

    -- Score based on consistency (lower stdDev = more perfect)
    -- Perfect circle has stdDev near 0
    local consistency = math.max(0, 100 - (stdDev / avgRadius) * 200)

    -- Check if drawing is somewhat closed (start and end points close)
    local closeness = 0
    if #points >= 2 then
        local startPoint = points[1]
        local endPoint = points[#points]
        local gap = distance(startPoint.x, startPoint.y, endPoint.x, endPoint.y)
        closeness = math.max(0, 100 - (gap / avgRadius) * 100)
    end

    -- Final score (weighted average)
    local finalScore = math.floor((consistency * 0.7 + closeness * 0.3))
    finalScore = math.max(0, math.min(100, finalScore))

    -- Generate feedback message
    local message = ""
    if finalScore >= 90 then
        message = "PERFECT! You're a circle master!"
    elseif finalScore >= 75 then
        message = "Great circle! Very round!"
    elseif finalScore >= 50 then
        message = "Not bad! Keep practicing."
    elseif finalScore >= 25 then
        message = "Looks more like an egg..."
    else
        message = "That's... abstract art?"
    end

    return finalScore, message
end

-- Get color based on score
local function getScoreColor(score)
    if score >= 75 then return COLOR_GOOD
    elseif score >= 50 then return COLOR_OKAY
    elseif score >= 25 then return COLOR_BAD
    else return COLOR_POOR
    end
end

-- Server-side networking
if SERVER then
    -- Net message to broadcast drawing to all clients
    net.receive("submitDrawing", function(len, ply)
        local points = net.readTable()
        local score = net.readFloat()
        local playerName = ply:getName()

        -- Update player's best score
        if not scores[playerName] or score > scores[playerName] then
            scores[playerName] = score
        end

        -- Broadcast to all clients
        net.start("receiveDrawing")
        net.writeString(playerName)
        net.writeTable(points)
        net.writeFloat(score)
        net.send()
    end)

    -- Send leaderboard on request
    net.receive("requestLeaderboard", function(len, ply)
        net.start("receiveLeaderboard")
        net.writeTable(scores)
        net.send(ply)
    end)
end

-- Client-side rendering and input
if CLIENT then
    -- Request leaderboard on spawn
    hook.add("ClientInitialized", "requestLeaderboard", function()
        timer.simple(1, function()
            net.start("requestLeaderboard")
            net.send()
        end)
    end)

    -- Receive drawing from server
    net.receive("receiveDrawing", function()
        local playerName = net.readString()
        local points = net.readTable()
        local score = net.readFloat()

        -- Add to active drawings
        table.insert(drawings, {
            playerName = playerName,
            points = points,
            score = score,
            time = timer.curtime(),
            color = getScoreColor(score)
        })
    end)

    -- Receive leaderboard update
    net.receive("receiveLeaderboard", function()
        scores = net.readTable()
    end)

    -- Input handling for drawing
    hook.add("inputPressed", "startDrawing", function(button)
        if button == MOUSE.MOUSE1 and render.cursorVisible() then
            isDrawing = true
            currentDrawing = {}
            startTime = timer.curtime()
        end
    end)

    hook.add("inputReleased", "endDrawing", function(button)
        if button == MOUSE.MOUSE1 and isDrawing then
            isDrawing = false

            if #currentDrawing >= MIN_POINTS then
                -- Evaluate the circle
                local score, message = evaluateCircle(currentDrawing)

                -- Send to server
                net.start("submitDrawing")
                net.writeTable(currentDrawing)
                net.writeFloat(score)
                net.send()

                -- Show local feedback
                print(message .. " Score: " .. score)

                -- Add to local drawings
                table.insert(drawings, {
                    playerName = player():getName(),
                    points = currentDrawing,
                    score = score,
                    time = timer.curtime(),
                    color = getScoreColor(score)
                })
            end

            currentDrawing = {}
        end
    end)

    -- Render loop
    hook.add("render", "drawGame", function()
        -- Clear background
        render.setColor(COLOR_BACKGROUND)
        render.drawRect(0, 0, CANVAS_SIZE, CANVAS_SIZE)

        -- Draw grid
        render.setColor(COLOR_GRID)
        for i = 0, CANVAS_SIZE, 64 do
            render.drawLine(i, 0, i, CANVAS_SIZE)
            render.drawLine(0, i, CANVAS_SIZE, i)
        end

        -- Draw center circle guide
        render.setColor(Color(100, 100, 150, 50))
        render.drawCircle(CANVAS_SIZE/2, CANVAS_SIZE/2, 150)

        -- Draw instructions
        render.setColor(Color(255, 255, 255))
        render.drawSimpleText(10, 10, "PERFECT CIRCLE CHALLENGE", "DermaLarge")
        render.drawSimpleText(10, 40, "Click and drag to draw a circle!", "DermaDefault")

        -- Update and draw active drawings from other players
        local currentTime = timer.curtime()
        for i = #drawings, 1, -1 do
            local drawing = drawings[i]

            -- Remove old drawings
            if currentTime - drawing.time > DRAW_TIME then
                table.remove(drawings, i)
            else
                -- Fade out over time
                local age = currentTime - drawing.time
                local alpha = math.floor(255 * (1 - age / DRAW_TIME))

                -- Draw the points
                render.setColor(Color(drawing.color.r, drawing.color.g, drawing.color.b, alpha))
                for j = 1, #drawing.points - 1 do
                    local p1 = drawing.points[j]
                    local p2 = drawing.points[j + 1]
                    render.drawLine(p1.x, p1.y, p2.x, p2.y)
                end

                -- Draw score label
                if #drawing.points > 0 then
                    local centerX = drawing.points[1].x
                    local centerY = drawing.points[1].y
                    render.setColor(Color(255, 255, 255, alpha))
                    render.drawSimpleText(centerX, centerY - 20, drawing.playerName, "DermaDefault")
                    render.drawSimpleText(centerX, centerY, "Score: " .. math.floor(drawing.score), "DermaDefault")
                end
            end
        end

        -- Draw current drawing in progress
        if isDrawing and #currentDrawing > 0 then
            render.setColor(Color(255, 255, 255))
            for i = 1, #currentDrawing - 1 do
                local p1 = currentDrawing[i]
                local p2 = currentDrawing[i + 1]
                render.drawLine(p1.x, p1.y, p2.x, p2.y)
            end

            -- Show point count
            render.drawSimpleText(10, 70, "Points: " .. #currentDrawing, "DermaDefault")
        end

        -- Draw leaderboard
        local y = CANVAS_SIZE - 200
        render.setColor(Color(0, 0, 0, 180))
        render.drawRect(10, y - 10, 220, 200)

        render.setColor(Color(255, 255, 255))
        render.drawSimpleText(15, y, "LEADERBOARD", "DermaDefault")

        -- Sort scores
        local sortedScores = {}
        for name, score in pairs(scores) do
            table.insert(sortedScores, {name = name, score = score})
        end
        table.sort(sortedScores, function(a, b) return a.score > b.score end)

        -- Display top 5
        y = y + 25
        for i = 1, math.min(5, #sortedScores) do
            local entry = sortedScores[i]
            local color = getScoreColor(entry.score)
            render.setColor(color)
            render.drawSimpleText(15, y, i .. ". " .. entry.name .. ": " .. math.floor(entry.score), "DermaDefault")
            y = y + 20
        end

        if #sortedScores == 0 then
            render.setColor(Color(150, 150, 150))
            render.drawSimpleText(15, y, "No scores yet!", "DermaDefault")
        end
    end)

    -- Track mouse position while drawing
    hook.add("think", "captureDrawing", function()
        if isDrawing and render.cursorVisible() then
            local cursorX, cursorY = render.cursorPos()

            -- Only add point if moved enough (smoothing)
            if #currentDrawing == 0 then
                table.insert(currentDrawing, {x = cursorX, y = cursorY})
            else
                local lastPoint = currentDrawing[#currentDrawing]
                local dist = distance(lastPoint.x, lastPoint.y, cursorX, cursorY)

                -- Add point if moved at least 3 pixels
                if dist > 3 then
                    table.insert(currentDrawing, {x = cursorX, y = cursorY})
                end
            end
        end
    end)

    -- Enable screen interaction
    render.createRenderTarget("circle_game")

    hook.add("renderOffscreen", "setupRT", function()
        render.selectRenderTarget("circle_game")
    end)

    hook.add("drawhud", "showCursor", function()
        -- This makes cursor visible when looking at the screen
    end)
end

-- Initialize
if CLIENT then
    -- Set up screen rendering
    enableHud(nil, true)
    render.createRenderTarget("screen")
end

print("Perfect Circle Challenge loaded! Look at a Starfall Screen to play!")
