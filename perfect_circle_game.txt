--[[
    Perfect Circle Challenge
    A multiplayer game where players compete to draw the most perfect circle!

    SETUP:
    1. Spawn this Starfall Processor chip
    2. Spawn a Starfall Screen
    3. Link the screen to this chip using the Starfall tool
    4. Look at the screen and click to draw!

    HOW TO PLAY:
    - Click and hold on the screen to draw
    - Draw a circle with your mouse
    - Release to submit and get scored
    - Try to draw the most perfect circle!
]]

if SERVER then
    -- Server-side: Network handling
    local scores = {} -- {steamID = {name, score}}

    net.receive("submitDrawing", function(len, ply)
        local points = net.readTable()
        local score = net.readFloat()
        local playerName = ply:getName()
        local steamID = ply:getSteamID()

        -- Update best score
        if not scores[steamID] or score > scores[steamID].score then
            scores[steamID] = {
                name = playerName,
                score = score
            }
        end

        -- Broadcast to all clients
        net.start("receiveDrawing")
        net.writeString(playerName)
        net.writeTable(points)
        net.writeFloat(score)
        net.send()
    end)

    net.receive("requestScores", function(len, ply)
        net.start("receiveScores")
        net.writeTable(scores)
        net.send(ply)
    end)
end

if CLIENT then
    -- Configuration
    local DRAW_TIME = 5
    local MIN_POINTS = 15
    local CANVAS_WIDTH = 512  -- Standard Starfall screen resolution
    local CANVAS_HEIGHT = 512

    -- Game state
    local playerDrawings = {}
    local playerScores = {}
    local myCurrentPoints = {}
    local isDrawing = false
    local lastMousePos = {x = 0, y = 0}

    -- Colors - Modern dark theme
    local COLOR_BG_DARK = Color(10, 10, 18)
    local COLOR_ACCENT = Color(100, 120, 255)
    local COLOR_ACCENT_DIM = Color(60, 80, 180)
    local COLOR_TEXT = Color(220, 220, 230)
    local COLOR_TEXT_DIM = Color(140, 140, 160)
    local COLOR_GRID = Color(30, 30, 45, 80)
    local COLOR_PERFECT = Color(50, 255, 150)
    local COLOR_GOOD = Color(100, 200, 255)
    local COLOR_OKAY = Color(255, 200, 50)
    local COLOR_BAD = Color(255, 100, 100)

    -- Helper: Calculate distance
    local function distance(x1, y1, x2, y2)
        local dx = x2 - x1
        local dy = y2 - y1
        return math.sqrt(dx * dx + dy * dy)
    end

    -- Evaluate circle perfection
    local function evaluateCircle(points)
        if #points < MIN_POINTS then
            return 0, "Too few points!"
        end

        local centerX, centerY = 0, 0
        for i = 1, #points do
            centerX = centerX + points[i].x
            centerY = centerY + points[i].y
        end
        centerX = centerX / #points
        centerY = centerY / #points

        local avgRadius = 0
        for i = 1, #points do
            avgRadius = avgRadius + distance(centerX, centerY, points[i].x, points[i].y)
        end
        avgRadius = avgRadius / #points

        if avgRadius < 20 then
            return 0, "Circle too small!"
        end

        local variance = 0
        for i = 1, #points do
            local radius = distance(centerX, centerY, points[i].x, points[i].y)
            local diff = radius - avgRadius
            variance = variance + (diff * diff)
        end
        variance = variance / #points
        local stdDev = math.sqrt(variance)

        local consistency = math.max(0, 100 - (stdDev / avgRadius) * 200)

        local closeness = 0
        if #points >= 2 then
            local gap = distance(points[1].x, points[1].y, points[#points].x, points[#points].y)
            closeness = math.max(0, 100 - (gap / avgRadius) * 100)
        end

        local finalScore = math.floor((consistency * 0.7 + closeness * 0.3))
        finalScore = math.max(0, math.min(100, finalScore))

        local message = ""
        if finalScore >= 95 then
            message = "FLAWLESS!"
        elseif finalScore >= 85 then
            message = "Nearly Perfect!"
        elseif finalScore >= 70 then
            message = "Great!"
        elseif finalScore >= 50 then
            message = "Not Bad!"
        elseif finalScore >= 30 then
            message = "Keep Trying..."
        else
            message = "Abstract?"
        end

        return finalScore, message
    end

    -- Get color for score
    local function getScoreColor(score)
        if score >= 85 then return COLOR_PERFECT
        elseif score >= 70 then return COLOR_GOOD
        elseif score >= 50 then return COLOR_OKAY
        else return COLOR_BAD
        end
    end

    -- Network receivers
    net.receive("receiveDrawing", function()
        local playerName = net.readString()
        local points = net.readTable()
        local score = net.readFloat()

        table.insert(playerDrawings, {
            playerName = playerName,
            points = points,
            score = score,
            message = evaluateCircle(points),
            time = timer.curtime(),
            color = getScoreColor(score)
        })
    end)

    net.receive("receiveScores", function()
        playerScores = net.readTable()
    end)

    -- Request initial scores
    timer.simple(0.5, function()
        net.start("requestScores")
        net.send()
    end)

    -- Input handling
    hook.add("inputPressed", "startDrawing", function(button)
        if button == MOUSE.MOUSE1 then
            isDrawing = true
            myCurrentPoints = {}
        end
    end)

    hook.add("inputReleased", "endDrawing", function(button)
        if button == MOUSE.MOUSE1 and isDrawing then
            isDrawing = false

            if #myCurrentPoints >= MIN_POINTS then
                local score, message = evaluateCircle(myCurrentPoints)

                -- Send to server
                net.start("submitDrawing")
                net.writeTable(myCurrentPoints)
                net.writeFloat(score)
                net.send()

                -- Add locally
                table.insert(playerDrawings, {
                    playerName = player():getName(),
                    points = myCurrentPoints,
                    score = score,
                    message = message,
                    time = timer.curtime(),
                    color = getScoreColor(score)
                })

                -- Update local score
                local steamID = player():getSteamID()
                if not playerScores[steamID] or score > playerScores[steamID].score then
                    playerScores[steamID] = {
                        name = player():getName(),
                        score = score
                    }
                end

                print(message .. " Score: " .. score)
            else
                print("Draw more! Need at least " .. MIN_POINTS .. " points")
            end

            myCurrentPoints = {}
        end
    end)

    -- Main render hook (runs when screen is linked)
    hook.add("render", "drawGame", function()
        -- Track mouse while drawing (must be in render hook)
        if isDrawing then
            local x, y = render.cursorPos()
            if x and y then
                x = math.clamp(x, 0, CANVAS_WIDTH)
                y = math.clamp(y, 0, CANVAS_HEIGHT)

                if #myCurrentPoints == 0 then
                    table.insert(myCurrentPoints, {x = x, y = y})
                    lastMousePos = {x = x, y = y}
                else
                    local dist = distance(lastMousePos.x, lastMousePos.y, x, y)
                    if dist > 3 then
                        table.insert(myCurrentPoints, {x = x, y = y})
                        lastMousePos = {x = x, y = y}
                    end
                end
            end
        end
        -- Background
        render.setColor(COLOR_BG_DARK)
        render.drawRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT)

        -- Grid
        render.setColor(COLOR_GRID)
        for i = 0, CANVAS_WIDTH, 64 do
            render.drawLine(i, 0, i, CANVAS_HEIGHT)
        end
        for i = 0, CANVAS_HEIGHT, 64 do
            render.drawLine(0, i, CANVAS_WIDTH, i)
        end

        -- Center guide circle
        render.setColor(Color(COLOR_ACCENT.r, COLOR_ACCENT.g, COLOR_ACCENT.b, 20))
        render.drawCircle(CANVAS_WIDTH/2, CANVAS_HEIGHT/2, 120)

        -- Header
        render.setColor(Color(20, 20, 35, 230))
        render.drawRect(0, 0, CANVAS_WIDTH, 80)

        render.setColor(COLOR_ACCENT)
        render.setFont("DermaLarge")
        render.drawText(CANVAS_WIDTH/2, 15, "PERFECT CIRCLE", 1)

        render.setColor(COLOR_TEXT_DIM)
        render.setFont("DermaDefault")
        render.drawText(CANVAS_WIDTH/2, 45, "Click and drag to draw!", 1)

        -- Drawing status
        if isDrawing then
            render.setColor(COLOR_ACCENT)
            render.drawText(CANVAS_WIDTH/2, 65, "Drawing: " .. #myCurrentPoints .. " pts", 1)
        end

        -- Draw submitted circles
        local currentTime = timer.curtime()
        for i = #playerDrawings, 1, -1 do
            local drawing = playerDrawings[i]

            if currentTime - drawing.time > DRAW_TIME then
                table.remove(playerDrawings, i)
            else
                local age = currentTime - drawing.time
                local alpha = math.floor(255 * (1 - age / DRAW_TIME))
                local col = drawing.color

                -- Main line
                render.setColor(Color(col.r, col.g, col.b, alpha))
                for j = 1, #drawing.points - 1 do
                    local p1 = drawing.points[j]
                    local p2 = drawing.points[j + 1]
                    render.drawLine(p1.x, p1.y, p2.x, p2.y)
                end

                -- Score label
                if #drawing.points > 0 then
                    local centerX, centerY = 0, 0
                    for _, p in ipairs(drawing.points) do
                        centerX = centerX + p.x
                        centerY = centerY + p.y
                    end
                    centerX = centerX / #drawing.points
                    centerY = centerY / #drawing.points

                    render.setColor(Color(20, 20, 30, alpha * 0.9))
                    render.drawRect(centerX - 40, centerY - 25, 80, 40)

                    render.setColor(Color(255, 255, 255, alpha))
                    render.setFont("DermaDefault")
                    render.drawText(centerX, centerY - 15, drawing.playerName, 1)

                    render.setColor(Color(col.r, col.g, col.b, alpha))
                    render.setFont("DermaDefaultBold")
                    render.drawText(centerX, centerY, math.floor(drawing.score), 1)

                    render.setFont("DermaDefault")
                    render.setColor(Color(200, 200, 200, alpha))
                    render.drawText(centerX, centerY + 12, drawing.message, 1)
                end
            end
        end

        -- Current drawing
        if isDrawing and #myCurrentPoints > 0 then
            render.setColor(Color(255, 255, 255, 255))
            for i = 1, #myCurrentPoints - 1 do
                local p1 = myCurrentPoints[i]
                local p2 = myCurrentPoints[i + 1]
                render.drawLine(p1.x, p1.y, p2.x, p2.y)
            end

            -- Start point marker
            if #myCurrentPoints > 0 then
                local start = myCurrentPoints[1]
                render.setColor(COLOR_ACCENT)
                render.drawCircle(start.x, start.y, 4)
            end
        end

        -- Leaderboard
        local boardWidth = 200
        local boardX = CANVAS_WIDTH - boardWidth - 10
        local boardY = 90

        render.setColor(Color(20, 20, 35, 240))
        render.drawRect(boardX, boardY, boardWidth, 40)

        render.setColor(COLOR_ACCENT)
        render.setFont("DermaDefaultBold")
        render.drawText(boardX + 10, boardY + 12, "LEADERBOARD", 0)

        local sortedScores = {}
        for steamID, data in pairs(playerScores) do
            table.insert(sortedScores, data)
        end
        table.sort(sortedScores, function(a, b) return a.score > b.score end)

        local entryY = boardY + 40
        for i = 1, math.min(5, #sortedScores) do
            local entry = sortedScores[i]
            local entryHeight = 35

            render.setColor(Color(25, 25, 40, 230))
            render.drawRect(boardX, entryY, boardWidth, entryHeight)

            -- Rank
            local rankColor = i == 1 and Color(255, 215, 0) or (i == 2 and Color(192, 192, 192) or Color(205, 127, 50))
            if i <= 3 then
                render.setColor(rankColor)
                render.drawCircle(boardX + 20, entryY + 17, 12)
                render.setColor(Color(20, 20, 30))
                render.setFont("DermaDefaultBold")
                render.drawText(boardX + 20, entryY + 11, i, 1)
            else
                render.setColor(COLOR_TEXT_DIM)
                render.setFont("DermaDefault")
                render.drawText(boardX + 20, entryY + 14, i, 1)
            end

            -- Name
            render.setColor(COLOR_TEXT)
            render.setFont("DermaDefault")
            local nameText = entry.name
            if #nameText > 12 then
                nameText = string.sub(nameText, 1, 12) .. "..."
            end
            render.drawText(boardX + 40, entryY + 8, nameText, 0)

            -- Score
            local scoreColor = getScoreColor(entry.score)
            render.setColor(scoreColor)
            render.setFont("DermaDefaultBold")
            render.drawText(boardX + 40, entryY + 20, math.floor(entry.score) .. " pts", 0)

            entryY = entryY + entryHeight + 5
        end

        if #sortedScores == 0 then
            render.setColor(COLOR_TEXT_DIM)
            render.setFont("DermaDefault")
            render.drawText(boardX + boardWidth/2, boardY + 60, "No scores yet!", 1)
        end
    end)

    print("Perfect Circle Challenge loaded!")
end
