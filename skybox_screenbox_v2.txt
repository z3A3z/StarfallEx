--@name Skybox Screen Box v2 (Screen Entities)
--@author
--@shared

--[[
    NUMPAD 1: Set first corner point
    NUMPAD 2: Set second corner point
    NUMPAD 3: Create screen box with skybox

    This version spawns actual starfall_screen entities (like manual workflow)
    but automates the process with numpad keys.
]]

if SERVER then
    local point1 = nil
    local point2 = nil
    local screenEnts = {}

    -- Net message to receive points from client
    hook.add("ClientInitialized", "screenbox_init", function(ply)
        if ply == owner() then
            -- Client initialized
        end
    end)

    -- Clear screens
    local function clearScreens()
        for _, screen in ipairs(screenEnts) do
            if screen:isValid() then
                screen:remove()
            end
        end
        screenEnts = {}
    end

    -- Create screen box
    net.receive("screenbox_create", function(len, ply)
        if ply ~= owner() then return end

        point1 = net.readVector()
        point2 = net.readVector()

        if not point1 or not point2 then return end

        clearScreens()

        -- Calculate bounds
        local minPos = Vector(
            math.min(point1.x, point2.x),
            math.min(point1.y, point2.y),
            math.min(point1.z, point2.z)
        )

        local maxPos = Vector(
            math.max(point1.x, point2.x),
            math.max(point1.y, point2.y),
            math.max(point1.z, point2.z)
        )

        local center = (minPos + maxPos) / 2
        local size = maxPos - minPos
        local width = size.x
        local height = size.y
        local depth = size.z

        -- Screen configurations (facing inward)
        local screens = {
            {pos = Vector(0, height/2, 0),    ang = Angle(0, 180, 90),  scale = Vector(width/48, depth/48, 1)},   -- Front
            {pos = Vector(width/2, 0, 0),     ang = Angle(0, -90, 90),  scale = Vector(height/48, depth/48, 1)},  -- Right
            {pos = Vector(0, -height/2, 0),   ang = Angle(0, 0, 90),    scale = Vector(width/48, depth/48, 1)},   -- Back
            {pos = Vector(-width/2, 0, 0),    ang = Angle(0, 90, 90),   scale = Vector(height/48, depth/48, 1)},  -- Left
            {pos = Vector(0, 0, depth/2),     ang = Angle(0, 180, 180), scale = Vector(width/48, height/48, 1)},  -- Top
            {pos = Vector(0, 0, -depth/2),    ang = Angle(0, 180, 0),   scale = Vector(width/48, height/48, 1)}   -- Bottom
        }

        -- Spawn screen entities
        for i, screen in ipairs(screens) do
            local ent = prop.createComponent(
                center + screen.pos,
                screen.ang,
                "starfall_screen",
                "models/hunter/plates/plate1x1.mdl",
                true
            )

            -- Scale the screen entity (note: might need adjustment per model)
            ent:setScale(screen.scale)

            -- Link screen to this chip
            timer.simple(0.1 * i, function()
                if ent:isValid() then
                    ent:linkComponent(chip())
                end
            end)

            table.insert(screenEnts, ent)
        end

        print("[Screen Box] Created " .. #screenEnts .. " screen entities")
    end)

else -- CLIENT
    local point1 = nil
    local point2 = nil
    local screensActive = false

    -- Skybox rendering code (same as working version)
    local scale = 200000
    local tex = {
        render.createMaterial("https://raw.githubusercontent.com/ax255/public-stuff/refs/heads/main/star-skybox/pz.png"),
        render.createMaterial("https://raw.githubusercontent.com/ax255/public-stuff/refs/heads/main/star-skybox/nx.png"),
        render.createMaterial("https://raw.githubusercontent.com/ax255/public-stuff/refs/heads/main/star-skybox/nz.png"),
        render.createMaterial("https://raw.githubusercontent.com/ax255/public-stuff/refs/heads/main/star-skybox/px.png"),
        render.createMaterial("https://raw.githubusercontent.com/ax255/public-stuff/refs/heads/main/star-skybox/py.png"),
        render.createMaterial("https://raw.githubusercontent.com/ax255/public-stuff/refs/heads/main/star-skybox/ny.png"),
    }

    local m = {}
    local angles = {
        Angle(0, 0, -90),
        Angle(0, 90, -90),
        Angle(0, 180, -90),
        Angle(0, -90, -90),
        Angle(0, 0, 0),
        Angle(0, 0, 180)
    }

    local pos = chip():getPos() + Vector(0, 0, -scale/10)

    for i = 1, 6 do
        m[i] = Matrix()
        m[i]:translate(pos)
        m[i]:rotate(angles[i])
        m[i]:scale(Vector(scale, scale, scale))
        m[i]:translate(Vector(-0.5, -0.5, 0.5))
    end

    -- Render skybox (runs on linked screens)
    hook.add("render", "screenbox_skybox", function()
        if not screensActive then return end

        local val = math.abs(math.sin(timer.curtime() * 0.05))

        for i = 1, 6 do
            render.pushViewMatrix({type = "3D", zfar = 300000})
            render.pushMatrix(m[i], true)
            render.setMaterial(tex[i])
            render.setMaterialEffectBloom(tex[i], 0.5 + 0.6 * val, 0.5 + 0.6 * val, 0.5 + 0.6 * val, 0.5 + 0.6 * val)
            render.drawTexturedRectUV(0, 0, 1, 1, 0.0007, 0.0007, 0.9993, 0.9993)
            render.popMatrix()
            render.popViewMatrix()
        end
    end)

    -- Set point from trace
    local function setPoint(num)
        local trace = trace.line(
            owner():getEyePos(),
            owner():getEyePos() + owner():getEyeAngles():getForward() * 10000,
            owner()
        )

        if trace.Hit then
            if num == 1 then
                point1 = trace.HitPos
                print("[Screen Box] Point 1 set")
            elseif num == 2 then
                point2 = trace.HitPos
                print("[Screen Box] Point 2 set")
            end
        end
    end

    -- Create screen box
    local function createScreenBox()
        if not point1 or not point2 then
            print("[Screen Box] Error: Set both points first!")
            return
        end

        screensActive = true
        net.start("screenbox_create")
        net.writeVector(point1)
        net.writeVector(point2)
        net.send()
    end

    -- Key input
    hook.add("inputPressed", "screenbox_keys", function(key)
        if key == KEY.PAD_1 then
            setPoint(1)
        elseif key == KEY.PAD_2 then
            setPoint(2)
        elseif key == KEY.PAD_3 then
            createScreenBox()
        end
    end)

    -- HUD
    hook.add("postdrawhud", "screenbox_hud", function()
        render.setColor(Color(255, 255, 255))
        render.drawSimpleText(10, 50, "SKYBOX SCREEN BOX v2", "DermaLarge")
        render.drawSimpleText(10, 80, "Numpad 1: Point 1", "DermaDefault")
        render.drawSimpleText(10, 100, "Numpad 2: Point 2", "DermaDefault")
        render.drawSimpleText(10, 120, "Numpad 3: Create", "DermaDefault")

        render.setColor(point1 and Color(0, 255, 0) or Color(255, 0, 0))
        render.drawSimpleText(10, 150, "Point 1: " .. (point1 and "SET" or "NOT SET"), "DermaDefault")

        render.setColor(point2 and Color(0, 255, 0) or Color(255, 0, 0))
        render.drawSimpleText(10, 170, "Point 2: " .. (point2 and "SET" or "NOT SET"), "DermaDefault")

        render.setColor(Color(255, 255, 255))
        render.drawSimpleText(10, 190, "Active: " .. (screensActive and "YES" or "NO"), "DermaDefault")
    end)

    -- 3D markers
    hook.add("postdrawopaquererenderables", "screenbox_markers", function()
        if point1 then
            render.draw3DBox(point1, Angle(0, 0, 0), Vector(-5, -5, -5), Vector(5, 5, 5), Color(0, 255, 0, 200))
        end

        if point2 then
            render.draw3DBox(point2, Angle(0, 0, 0), Vector(-5, -5, -5), Vector(5, 5, 5), Color(255, 0, 0, 200))
        end

        if point1 and point2 then
            render.draw3DLine(point1, point2, Color(255, 255, 0, 200))
        end
    end)
end
