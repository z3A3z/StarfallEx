-- UI Constants
local CANVAS_WIDTH = 512
local CANVAS_HEIGHT = 512
local COOKIE_SIZE = 80
local COOKIE_Y = 200

-- Colors
local COLOR_BG = Color(25, 25, 40)
local COLOR_BG_LIGHT = Color(35, 35, 55)
local COLOR_ACCENT = Color(220, 160, 70)
local COLOR_ACCENT_BRIGHT = Color(255, 200, 100)
local COLOR_TEXT = Color(240, 240, 250)
local COLOR_TEXT_DIM = Color(160, 160, 180)
local COLOR_BUTTON = Color(50, 50, 70)
local COLOR_BUTTON_HOVER = Color(60, 60, 85)
local COLOR_BUTTON_DISABLED = Color(30, 30, 40)
local COLOR_GREEN = Color(80, 220, 100)
local COLOR_RED = Color(220, 80, 80)
local COLOR_GOLDEN = Color(255, 215, 0)

-- Animation helpers
local cookieRotation = 0
local particleTime = 0

-- Helper: Check if point in rect
local function pointInRect(x, y, rx, ry, rw, rh)
    return x >= rx and x <= rx + rw and y >= ry and y <= ry + rh
end

-- Helper: Draw button
local function drawButton(x, y, w, h, text, canAfford, count)
    local color = canAfford and COLOR_BUTTON or COLOR_BUTTON_DISABLED

    -- Background
    render.setColor(color)
    render.drawRect(x, y, w, h)

    -- Border
    local borderColor = canAfford and COLOR_ACCENT or COLOR_TEXT_DIM
    render.setColor(borderColor)
    render.drawRect(x, y, w, 2)
    render.drawRect(x, y + h - 2, w, 2)
    render.drawRect(x, y, 2, h)
    render.drawRect(x + w - 2, y, 2, h)

    -- Text
    render.setColor(canAfford and COLOR_TEXT or COLOR_TEXT_DIM)
    render.setFont("DermaDefault")
    render.drawText(x + 5, y + 5, text, 0)

    if count then
        render.setColor(COLOR_ACCENT_BRIGHT)
        render.setFont("DermaDefaultBold")
        render.drawText(x + w - 5, y + 5, count, 2)
    end
end

-- Helper: Draw upgrade button
local function drawUpgradeButton(x, y, size, name, owned, canAfford)
    local color = owned and COLOR_GREEN or (canAfford and COLOR_BUTTON or COLOR_BUTTON_DISABLED)

    render.setColor(color)
    render.drawRect(x, y, size, size)

    -- Border
    local borderColor = owned and COLOR_GREEN or (canAfford and COLOR_ACCENT or COLOR_TEXT_DIM)
    render.setColor(borderColor)
    render.drawRect(x, y, size, 2)
    render.drawRect(x, y + size - 2, size, 2)
    render.drawRect(x, y, 2, size)
    render.drawRect(x + size - 2, y, 2, size)

    -- Checkmark if owned
    if owned then
        render.setColor(COLOR_TEXT)
        render.setFont("DermaLarge")
        render.drawText(x + size/2, y + size/2 - 10, "✓", 1)
    end
end

-- Draw the giant cookie in the center
local function drawCookie(scale, rotation)
    local centerX = CANVAS_WIDTH / 2
    local centerY = COOKIE_Y
    local size = COOKIE_SIZE * scale

    -- Cookie shadow
    render.setColor(Color(0, 0, 0, 100))
    render.drawFilledCircle(centerX + 5, centerY + 5, size)

    -- Main cookie (layered circles for depth)
    render.setColor(Color(210, 140, 70))
    render.drawFilledCircle(centerX, centerY, size)

    render.setColor(Color(190, 120, 60))
    render.drawFilledCircle(centerX, centerY, size * 0.9)

    render.setColor(Color(210, 140, 70))
    render.drawFilledCircle(centerX, centerY, size * 0.8)

    -- Chocolate chips (static positions based on rotation)
    local chips = {
        {0.3, 0.2}, {-0.4, 0.1}, {0.2, -0.3},
        {-0.2, -0.4}, {0.5, -0.1}, {-0.3, 0.4},
        {0.1, 0.5}, {-0.5, -0.2}
    }

    render.setColor(Color(80, 50, 30))
    for _, chip in ipairs(chips) do
        local cx = centerX + chip[1] * size * 0.6
        local cy = centerY + chip[2] * size * 0.6
        render.drawFilledCircle(cx, cy, size * 0.15)
    end

    -- Highlight
    render.setColor(Color(255, 255, 255, 100))
    render.drawFilledCircle(centerX - size * 0.3, centerY - size * 0.3, size * 0.25)
end

-- Draw golden cookie
local function drawGoldenCookie(x, y, timeLeft)
    local pulse = 1.0 + math.sin(timer.curtime() * 8) * 0.1
    local size = 25 * pulse

    -- Glow effect
    render.setColor(Color(255, 215, 0, 100))
    render.drawFilledCircle(x, y, size * 1.5)

    -- Main golden cookie
    render.setColor(COLOR_GOLDEN)
    render.drawFilledCircle(x, y, size)

    render.setColor(Color(255, 235, 100))
    render.drawFilledCircle(x, y, size * 0.7)

    -- Sparkle
    render.setColor(Color(255, 255, 255))
    render.drawFilledCircle(x - size * 0.3, y - size * 0.3, size * 0.2)

    -- Timer
    render.setColor(COLOR_TEXT)
    render.setFont("DermaDefault")
    render.drawText(x, y + size + 10, string.format("%.1fs", timeLeft), 1)
end

-- Draw click animations
local function drawClickAnimations(animations)
    local currentTime = timer.curtime()

    for i = #animations, 1, -1 do
        local anim = animations[i]
        local age = currentTime - anim.time

        if age > 1.0 then
            table.remove(animations, i)
        else
            local alpha = math.floor(255 * (1 - age))
            local yOffset = age * 50

            if anim.golden then
                render.setColor(Color(255, 215, 0, alpha))
            else
                render.setColor(Color(80, 220, 100, alpha))
            end

            render.setFont("DermaLarge")
            local text = "+" .. formatNumber(anim.amount)
            render.drawText(anim.x, anim.y - yOffset, text, 1)
        end
    end
end

-- Draw header with stats
local function drawHeader(gameState)
    -- Header background
    render.setColor(Color(20, 20, 35, 240))
    render.drawRect(0, 0, CANVAS_WIDTH, 100)

    -- Cookie count
    render.setColor(COLOR_ACCENT_BRIGHT)
    render.setFont("DermaLarge")
    render.drawText(CANVAS_WIDTH/2, 15, formatNumber(gameState.cookies), 1)

    render.setColor(COLOR_TEXT)
    render.setFont("DermaDefault")
    render.drawText(CANVAS_WIDTH/2, 40, "cookies", 1)

    -- Per second
    render.setColor(COLOR_GREEN)
    render.setFont("DermaDefaultBold")
    render.drawText(CANVAS_WIDTH/2, 60, "per second: " .. formatNumber(gameState.cookiesPerSecond), 1)

    -- Per click
    render.setColor(COLOR_ACCENT)
    render.setFont("DermaDefault")
    render.drawText(CANVAS_WIDTH/2, 80, "per click: " .. formatNumber(gameState.cookiesPerClick), 1)
end

-- Draw buildings panel
local function drawBuildings(gameState, scrollOffset)
    local panelX = 10
    local panelY = 300
    local panelWidth = CANVAS_WIDTH - 20
    local panelHeight = CANVAS_HEIGHT - panelY - 10

    -- Panel background
    render.setColor(COLOR_BG_LIGHT)
    render.drawRect(panelX, panelY, panelWidth, panelHeight)

    -- Title
    render.setColor(COLOR_ACCENT)
    render.setFont("DermaDefaultBold")
    render.drawText(panelX + 10, panelY + 5, "BUILDINGS", 0)

    -- Building buttons
    local buttonY = panelY + 30
    local visibleBuildings = math.min(4, #BUILDINGS)

    for i = 1, visibleBuildings do
        local building = BUILDINGS[i]
        local count = gameState.buildings[i] or 0
        local cost = getBuildingCost(i)
        local canAfford = gameState.cookies >= cost

        local btnX = panelX + 5
        local btnY = buttonY + (i - 1) * 45
        local btnW = panelWidth - 10
        local btnH = 40

        -- Button background
        local bgColor = canAfford and COLOR_BUTTON or COLOR_BUTTON_DISABLED
        render.setColor(bgColor)
        render.drawRect(btnX, btnY, btnW, btnH)

        -- Border
        local borderColor = canAfford and COLOR_ACCENT or COLOR_TEXT_DIM
        render.setColor(borderColor)
        render.drawRect(btnX, btnY, btnW, 2)
        render.drawRect(btnX, btnY + btnH - 2, btnW, 2)
        render.drawRect(btnX, btnY, 2, btnH)
        render.drawRect(btnX + btnW - 2, btnY, 2, btnH)

        -- Building info
        render.setColor(canAfford and COLOR_TEXT or COLOR_TEXT_DIM)
        render.setFont("DermaDefaultBold")
        render.drawText(btnX + 10, btnY + 5, building.name, 0)

        render.setFont("DermaDefault")
        render.drawText(btnX + 10, btnY + 23, formatNumber(cost), 0)

        -- Count owned
        if count > 0 then
            render.setColor(COLOR_ACCENT_BRIGHT)
            render.setFont("DermaLarge")
            render.drawText(btnX + btnW - 10, btnY + 10, count, 2)
        end

        -- CPS info
        render.setColor(COLOR_GREEN)
        render.setFont("DermaDefault")
        render.drawText(btnX + btnW - 10, btnY + 28, "+" .. formatNumber(building.cps), 2)
    end
end

-- Draw upgrades panel
local function drawUpgrades(gameState)
    local panelX = 10
    local panelY = 300
    local panelWidth = CANVAS_WIDTH - 20
    local panelHeight = CANVAS_HEIGHT - panelY - 10

    -- Panel background
    render.setColor(COLOR_BG_LIGHT)
    render.drawRect(panelX, panelY, panelWidth, panelHeight)

    -- Title
    render.setColor(COLOR_ACCENT)
    render.setFont("DermaDefaultBold")
    render.drawText(panelX + 10, panelY + 5, "UPGRADES", 0)

    -- Upgrade grid (4 per row)
    local upgradeSize = 60
    local spacing = 5
    local startY = panelY + 30
    local startX = panelX + 10
    local perRow = 7

    local shown = 0
    for i = 1, math.min(28, #UPGRADES) do
        local upgrade = UPGRADES[i]
        local owned = gameState.upgrades[i] or false
        local canAfford = gameState.cookies >= upgrade.cost

        -- Check if upgrade should be shown (requirements met)
        local shouldShow = owned or (upgrade.requires and upgrade.requires(gameState))

        if shouldShow then
            local col = shown % perRow
            local row = math.floor(shown / perRow)

            local x = startX + col * (upgradeSize + spacing)
            local y = startY + row * (upgradeSize + spacing)

            -- Only draw if in visible area
            if y + upgradeSize < panelY + panelHeight then
                -- Upgrade box
                local bgColor = owned and COLOR_GREEN or (canAfford and COLOR_BUTTON or COLOR_BUTTON_DISABLED)
                render.setColor(bgColor)
                render.drawRect(x, y, upgradeSize, upgradeSize)

                -- Border
                local borderColor = owned and COLOR_GREEN or (canAfford and COLOR_ACCENT or COLOR_TEXT_DIM)
                render.setColor(borderColor)
                render.drawRect(x, y, upgradeSize, 2)
                render.drawRect(x, y + upgradeSize - 2, upgradeSize, 2)
                render.drawRect(x, y, 2, upgradeSize)
                render.drawRect(x + upgradeSize - 2, y, 2, upgradeSize)

                -- Checkmark or cost
                if owned then
                    render.setColor(COLOR_TEXT)
                    render.setFont("DermaLarge")
                    render.drawText(x + upgradeSize/2, y + upgradeSize/2 - 10, "✓", 1)
                else
                    render.setColor(canAfford and COLOR_TEXT or COLOR_TEXT_DIM)
                    render.setFont("DermaDefault")
                    render.drawText(x + upgradeSize/2, y + 5, upgrade.name:sub(1, 8), 1)
                    render.drawText(x + upgradeSize/2, y + upgradeSize - 15, formatNumber(upgrade.cost), 1)
                end
            end

            shown = shown + 1
        end
    end

    if shown == 0 then
        render.setColor(COLOR_TEXT_DIM)
        render.setFont("DermaDefault")
        render.drawText(panelX + panelWidth/2, panelY + 60, "No upgrades available yet", 1)
    end
end

-- Draw tab selector
local function drawTabs(selectedTab)
    local tabY = 280
    local tabWidth = (CANVAS_WIDTH - 20) / 2
    local tabHeight = 25

    -- Buildings tab
    local buildingColor = selectedTab == "buildings" and COLOR_ACCENT or COLOR_BUTTON
    render.setColor(buildingColor)
    render.drawRect(10, tabY, tabWidth - 5, tabHeight)
    render.setColor(COLOR_TEXT)
    render.setFont("DermaDefaultBold")
    render.drawText(10 + tabWidth/2 - 5, tabY + 7, "BUILDINGS", 1)

    -- Upgrades tab
    local upgradeColor = selectedTab == "upgrades" and COLOR_ACCENT or COLOR_BUTTON
    render.setColor(upgradeColor)
    render.drawRect(15 + tabWidth, tabY, tabWidth - 5, tabHeight)
    render.setColor(COLOR_TEXT)
    render.drawText(15 + tabWidth + tabWidth/2 - 5, tabY + 7, "UPGRADES", 1)
end

-- Main render function
function renderGame(gameState, selectedTab, nearChip, clickAnimations, cookieScale, rotation)
    -- Background
    render.setColor(COLOR_BG)
    render.drawRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT)

    -- Header with stats
    drawHeader(gameState)

    -- Giant cookie
    cookieRotation = cookieRotation + 0.3
    drawCookie(cookieScale, cookieRotation)

    -- Golden cookie
    if gameState.goldenCookie and gameState.goldenCookie.active then
        local timeLeft = gameState.goldenCookie.expires - timer.curtime()
        if timeLeft > 0 then
            drawGoldenCookie(gameState.goldenCookie.x, gameState.goldenCookie.y, timeLeft)
        end
    end

    -- Click animations
    drawClickAnimations(clickAnimations)

    -- Tabs
    drawTabs(selectedTab)

    -- Panel content
    if selectedTab == "buildings" then
        drawBuildings(gameState, 0)
    else
        drawUpgrades(gameState)
    end

    -- Distance warning
    if not nearChip then
        render.setColor(Color(0, 0, 0, 200))
        render.drawRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT)

        render.setColor(COLOR_ACCENT)
        render.setFont("DermaLarge")
        render.drawText(CANVAS_WIDTH/2, CANVAS_HEIGHT/2 - 15, "TOO FAR", 1)

        render.setColor(COLOR_TEXT)
        render.setFont("DermaDefault")
        render.drawText(CANVAS_WIDTH/2, CANVAS_HEIGHT/2 + 15, "Get closer to the chip", 1)
    end
end

-- Handle UI clicks
function handleUIClick(mx, my, gameState, currentTab)
    -- Check tab clicks
    local tabY = 280
    local tabWidth = (CANVAS_WIDTH - 20) / 2
    local tabHeight = 25

    if pointInRect(mx, my, 10, tabY, tabWidth - 5, tabHeight) then
        selectedTab = "buildings"
        return "buildings"
    elseif pointInRect(mx, my, 15 + tabWidth, tabY, tabWidth - 5, tabHeight) then
        selectedTab = "upgrades"
        return "upgrades"
    end

    -- Check building clicks
    if currentTab == "buildings" then
        local panelY = 300
        local buttonY = panelY + 30
        local visibleBuildings = math.min(4, #BUILDINGS)

        for i = 1, visibleBuildings do
            local btnX = 15
            local btnY = buttonY + (i - 1) * 45
            local btnW = CANVAS_WIDTH - 30
            local btnH = 40

            if pointInRect(mx, my, btnX, btnY, btnW, btnH) then
                -- Try to buy building
                net.start("buyBuilding")
                net.writeUInt(i, 8)
                net.send()
                return currentTab
            end
        end
    end

    -- Check upgrade clicks
    if currentTab == "upgrades" then
        local panelY = 300
        local upgradeSize = 60
        local spacing = 5
        local startY = panelY + 30
        local startX = 20
        local perRow = 7

        local shown = 0
        for i = 1, math.min(28, #UPGRADES) do
            local upgrade = UPGRADES[i]
            local owned = gameState.upgrades[i] or false
            local shouldShow = owned or (upgrade.requires and upgrade.requires(gameState))

            if shouldShow then
                local col = shown % perRow
                local row = math.floor(shown / perRow)

                local x = startX + col * (upgradeSize + spacing)
                local y = startY + row * (upgradeSize + spacing)

                if pointInRect(mx, my, x, y, upgradeSize, upgradeSize) and not owned then
                    -- Try to buy upgrade
                    net.start("buyUpgrade")
                    net.writeUInt(i, 8)
                    net.send()
                    return currentTab
                end

                shown = shown + 1
            end
        end
    end

    return currentTab
end
