--[[
    Cookie Clicker Empire - Multiplayer Idle Game

    SETUP:
    1. Spawn this Starfall Processor chip
    2. Spawn a Starfall Screen
    3. Link them with the Starfall tool
    4. Start clicking cookies!

    FEATURES:
    - 12 different building types
    - 30+ upgrades
    - Golden cookie events
    - Achievements
    - Multiplayer - everyone controls the same empire!
    - Particle effects and animations
    - Prestige system for late game
]]

-- Load included files
require("cookie_clicker_data")

-- Shared game state structure
GAME = GAME or {
    cookies = 0,
    totalCookies = 0,
    cookiesPerSecond = 0,
    cookiesPerClick = 1,
    buildings = {},
    upgrades = {},
    goldenCookie = {active = false, x = 0, y = 0, expires = 0, multiplier = 1},
    stats = {
        clicks = 0,
        goldenClicks = 0,
        buildingsBought = 0
    }
}

if SERVER then
    -- Initialize building counts
    for i = 1, #BUILDINGS do
        GAME.buildings[i] = GAME.buildings[i] or 0
    end

    -- Calculate cookies per second
    local function calculateCPS()
        local cps = 0
        for i = 1, #BUILDINGS do
            local building = BUILDINGS[i]
            local count = GAME.buildings[i] or 0
            local multiplier = 1

            -- Apply upgrades
            for _, upgradeId in ipairs(building.upgradeIds or {}) do
                if GAME.upgrades[upgradeId] then
                    multiplier = multiplier * UPGRADES[upgradeId].multiplier
                end
            end

            cps = cps + (building.cps * count * multiplier)
        end
        GAME.cookiesPerSecond = cps
    end

    -- Calculate cookies per click
    local function calculateCPC()
        local cpc = 1
        -- Cursor buildings add to click power
        local cursors = GAME.buildings[1] or 0
        cpc = cpc + cursors * 0.1

        -- Click upgrades
        for id, owned in pairs(GAME.upgrades) do
            if owned and UPGRADES[id] and UPGRADES[id].clickBonus then
                cpc = cpc + UPGRADES[id].clickBonus
            end
        end

        GAME.cookiesPerClick = cpc
    end

    -- Passive cookie generation
    timer.create("cookieGen", 0.1, 0, function()
        if GAME.cookiesPerSecond > 0 then
            local gained = GAME.cookiesPerSecond * 0.1
            GAME.cookies = GAME.cookies + gained
            GAME.totalCookies = GAME.totalCookies + gained
        end
    end)

    -- Golden cookie spawner
    timer.create("goldenCookie", 30, 0, function()
        if not GAME.goldenCookie.active and math.random() < 0.3 then
            GAME.goldenCookie.active = true
            GAME.goldenCookie.x = math.random(100, 412)
            GAME.goldenCookie.y = math.random(100, 412)
            GAME.goldenCookie.expires = timer.curtime() + 10
            GAME.goldenCookie.multiplier = math.random(5, 15)

            -- Broadcast golden cookie
            net.start("goldenCookie")
            net.writeTable(GAME.goldenCookie)
            net.send()
        end
    end)

    -- Check golden cookie expiration
    hook.add("think", "goldenExpire", function()
        if GAME.goldenCookie.active and timer.curtime() > GAME.goldenCookie.expires then
            GAME.goldenCookie.active = false
        end
    end)

    -- Network: Click cookie
    net.receive("clickCookie", function(len, ply)
        local isGolden = net.readBool()

        if isGolden and GAME.goldenCookie.active then
            -- Golden cookie clicked
            local reward = GAME.cookiesPerSecond * GAME.goldenCookie.multiplier
            GAME.cookies = GAME.cookies + reward
            GAME.totalCookies = GAME.totalCookies + reward
            GAME.stats.goldenClicks = GAME.stats.goldenClicks + 1
            GAME.goldenCookie.active = false

            -- Broadcast golden collect
            net.start("goldenCollected")
            net.writeFloat(reward)
            net.send()
        else
            -- Regular cookie click
            GAME.cookies = GAME.cookies + GAME.cookiesPerClick
            GAME.totalCookies = GAME.totalCookies + GAME.cookiesPerClick
            GAME.stats.clicks = GAME.stats.clicks + 1

            -- Broadcast click
            net.start("cookieClicked")
            net.writeFloat(GAME.cookiesPerClick)
            net.send()
        end
    end)

    -- Network: Buy building
    net.receive("buyBuilding", function(len, ply)
        local buildingId = net.readUInt(8)
        local building = BUILDINGS[buildingId]

        if building then
            local currentCount = GAME.buildings[buildingId] or 0
            local cost = building.baseCost * math.pow(1.15, currentCount)

            if GAME.cookies >= cost then
                GAME.cookies = GAME.cookies - cost
                GAME.buildings[buildingId] = currentCount + 1
                GAME.stats.buildingsBought = GAME.stats.buildingsBought + 1

                calculateCPS()
                calculateCPC()

                -- Broadcast purchase
                net.start("buildingBought")
                net.writeUInt(buildingId, 8)
                net.writeFloat(cost)
                net.send()
            end
        end
    end)

    -- Network: Buy upgrade
    net.receive("buyUpgrade", function(len, ply)
        local upgradeId = net.readUInt(8)
        local upgrade = UPGRADES[upgradeId]

        if upgrade and not GAME.upgrades[upgradeId] then
            if GAME.cookies >= upgrade.cost then
                GAME.cookies = GAME.cookies - upgrade.cost
                GAME.upgrades[upgradeId] = true

                calculateCPS()
                calculateCPC()

                -- Broadcast purchase
                net.start("upgradeBought")
                net.writeUInt(upgradeId, 8)
                net.send()
            end
        end
    end)

    -- Network: Request game state
    net.receive("requestState", function(len, ply)
        net.start("gameState")
        net.writeTable(GAME)
        net.send(ply)
    end)

    -- Broadcast state updates every second
    timer.create("stateSync", 1, 0, function()
        net.start("syncState")
        net.writeFloat(GAME.cookies)
        net.writeFloat(GAME.cookiesPerSecond)
        net.writeFloat(GAME.cookiesPerClick)
        net.writeTable(GAME.buildings)
        net.send()
    end)
end

if CLIENT then
    -- Load UI functions
    require("cookie_clicker_ui")

    -- Local game state (synced from server)
    local gameState = {
        cookies = 0,
        totalCookies = 0,
        cookiesPerSecond = 0,
        cookiesPerClick = 1,
        buildings = {},
        upgrades = {},
        goldenCookie = {active = false},
        stats = {clicks = 0, goldenClicks = 0, buildingsBought = 0}
    }

    -- UI state
    local MAX_DISTANCE = 250
    local CANVAS_WIDTH = 512
    local CANVAS_HEIGHT = 512
    selectedTab = "buildings" -- "buildings" or "upgrades" (global for UI file)

    -- Animation state
    local clickAnimations = {} -- {x, y, amount, time}
    local cookieScale = 1.0
    local cookieRotation = 0

    -- Helper: Check if player is near chip
    local function isPlayerNear()
        local ply = player()
        local chipEnt = chip()
        if not ply or not chipEnt then return false end
        return ply:getPos():getDistance(chipEnt:getPos()) <= MAX_DISTANCE
    end

    -- Helper: Format large numbers
    function formatNumber(num)
        if num >= 1e12 then
            return string.format("%.2fT", num / 1e12)
        elseif num >= 1e9 then
            return string.format("%.2fB", num / 1e9)
        elseif num >= 1e6 then
            return string.format("%.2fM", num / 1e6)
        elseif num >= 1e3 then
            return string.format("%.2fK", num / 1e3)
        else
            return string.format("%.0f", num)
        end
    end

    -- Helper: Get building cost
    function getBuildingCost(buildingId)
        local building = BUILDINGS[buildingId]
        local count = gameState.buildings[buildingId] or 0
        return building.baseCost * math.pow(1.15, count)
    end

    -- Network receivers
    net.receive("gameState", function()
        gameState = net.readTable()
    end)

    net.receive("syncState", function()
        gameState.cookies = net.readFloat()
        gameState.cookiesPerSecond = net.readFloat()
        gameState.cookiesPerClick = net.readFloat()
        gameState.buildings = net.readTable()
    end)

    net.receive("cookieClicked", function()
        local amount = net.readFloat()
        cookieScale = 1.15

        -- Add click animation at cookie center
        table.insert(clickAnimations, {
            x = CANVAS_WIDTH / 2,
            y = 200,
            amount = amount,
            time = timer.curtime()
        })
    end)

    net.receive("goldenCookie", function()
        gameState.goldenCookie = net.readTable()
    end)

    net.receive("goldenCollected", function()
        local reward = net.readFloat()
        table.insert(clickAnimations, {
            x = gameState.goldenCookie.x,
            y = gameState.goldenCookie.y,
            amount = reward,
            time = timer.curtime(),
            golden = true
        })
        gameState.goldenCookie.active = false
    end)

    net.receive("buildingBought", function()
        local buildingId = net.readUInt(8)
        local cost = net.readFloat()
        gameState.buildings[buildingId] = (gameState.buildings[buildingId] or 0) + 1
    end)

    net.receive("upgradeBought", function()
        local upgradeId = net.readUInt(8)
        gameState.upgrades[upgradeId] = true
    end)

    -- Request initial state
    timer.simple(0.5, function()
        net.start("requestState")
        net.send()
    end)

    -- Input handling
    local clickPending = false
    local clickX, clickY = 0, 0

    hook.add("inputPressed", "handleClick", function(button)
        if button == MOUSE.MOUSE1 and isPlayerNear() then
            clickPending = true
        end
    end)

    hook.add("inputReleased", "handleRelease", function(button)
        if button == MOUSE.MOUSE1 then
            clickPending = false
        end
    end)

    -- Main render loop
    hook.add("render", "drawGame", function()
        local nearChip = isPlayerNear()
        local mx, my = render.cursorPos()

        if mx and my and clickPending and nearChip then
            clickPending = false
            clickX, clickY = mx, my

            -- Check golden cookie click
            if gameState.goldenCookie.active then
                local gx, gy = gameState.goldenCookie.x, gameState.goldenCookie.y
                local dist = math.sqrt((mx - gx)^2 + (my - gy)^2)
                if dist < 30 then
                    net.start("clickCookie")
                    net.writeBool(true)
                    net.send()
                    clickPending = false
                    return
                end
            end

            -- Check main cookie click (center, radius ~80)
            local cookieCenterX = CANVAS_WIDTH / 2
            local cookieCenterY = 200
            local dist = math.sqrt((mx - cookieCenterX)^2 + (my - cookieCenterY)^2)
            if dist < 80 then
                net.start("clickCookie")
                net.writeBool(false)
                net.send()
            else
                -- Check building/upgrade buttons
                local newTab = handleUIClick(mx, my, gameState, selectedTab)
                if newTab then
                    selectedTab = newTab
                end
            end
        end

        -- Animate cookie scale back to normal
        if cookieScale > 1.0 then
            cookieScale = cookieScale - 0.02
        end

        -- Render everything
        renderGame(gameState, selectedTab, nearChip, clickAnimations, cookieScale, cookieRotation)
    end)

    print("Cookie Clicker Empire loaded! Start clicking!")
end
