--@name Movement Recorder & Cloner
--@author
--@client

--[[
    Press NUM 1: Spawn a hologram clone that replays your recorded movements
    Press NUM 2: Remove the last spawned clone

    The code continuously records your position, angles, model, and animation.
    Clones will loop the recorded movements indefinitely.
]]

-- Recording data
local recording = {}
local isRecording = true
local recordingStartTime = timer.curtime()

-- Clone management
local clones = {}

-- Record player data every frame
hook.add("think", "record_movement", function()
    if not isRecording then return end

    local ply = owner()
    if not ply:isValid() then return end

    -- Get body angles (only yaw, keep pitch/roll at 0 for proper standing pose)
    local bodyAng = ply:getAngles()
    local bodyYaw = Angle(0, bodyAng.y, 0)

    -- Get the actual animation sequence and progress from the player
    local seqID = ply:getSequence()
    local seqName = ply:getSequenceName(seqID)
    local animProgress = ply:getAnimationProgress()

    local frame = {
        pos = ply:getPos(),
        ang = bodyYaw,
        seqName = seqName,
        animProgress = animProgress,
        model = ply:getModel(),
        time = timer.curtime() - recordingStartTime
    }

    table.insert(recording, frame)
end)

-- Spawn a clone that replays the recording
local function spawnClone()
    if #recording == 0 then
        print("[Clone] No recording data available!")
        return
    end

    -- Make a copy of the current recording
    local recordingCopy = {}
    for i, frame in ipairs(recording) do
        recordingCopy[i] = {
            pos = Vector(frame.pos.x, frame.pos.y, frame.pos.z),
            ang = Angle(frame.ang.p, frame.ang.y, frame.ang.r),
            seqName = frame.seqName,
            animProgress = frame.animProgress,
            model = frame.model,
            time = frame.time
        }
    end

    -- Get starting frame data
    local startFrame = recordingCopy[1]

    -- Create hologram
    local holo = holograms.create(
        startFrame.pos,
        startFrame.ang,
        startFrame.model,
        Vector(1, 1, 1)
    )

    if not holo then
        print("[Clone] Failed to create hologram!")
        return
    end

    -- Set initial animation using the actual sequence name and progress (rate 0 = no auto-advance)
    holo:setAnimation(startFrame.seqName, startFrame.animProgress, 0)

    -- Store clone data
    local cloneData = {
        holo = holo,
        recording = recordingCopy,
        startTime = timer.curtime(),
        currentFrame = 1,
        currentSeq = startFrame.seqName,
        totalFrames = #recordingCopy
    }

    table.insert(clones, cloneData)

    print("[Clone] Spawned clone #" .. #clones .. " with " .. #recordingCopy .. " frames (" .. string.format("%.1f", recordingCopy[#recordingCopy].time) .. " seconds)")

    -- Reset recording for next clone
    recording = {}
    recordingStartTime = timer.curtime()
    print("[Clone] Recording reset - starting new recording")
end

-- Remove the last spawned clone
local function removeLastClone()
    if #clones == 0 then
        print("[Clone] No clones to remove!")
        return
    end

    local lastClone = table.remove(clones)
    if lastClone.holo:isValid() then
        lastClone.holo:remove()
    end

    print("[Clone] Removed clone. " .. #clones .. " clones remaining.")
end

-- Update all clones
hook.add("think", "update_clones", function()
    for i = #clones, 1, -1 do
        local clone = clones[i]

        if not clone.holo:isValid() then
            table.remove(clones, i)
        else
            local elapsed = timer.curtime() - clone.startTime
            local totalDuration = clone.recording[clone.totalFrames].time

            -- Loop the recording
            if totalDuration > 0 then
                elapsed = elapsed % totalDuration
            end

            -- Find the appropriate frame based on elapsed time
            local targetFrame = 1
            for j = 1, clone.totalFrames do
                if clone.recording[j].time <= elapsed then
                    targetFrame = j
                else
                    break
                end
            end

            -- Apply the frame data
            local frame = clone.recording[targetFrame]
            clone.holo:setPos(frame.pos)
            clone.holo:setAngles(frame.ang)

            -- Update animation - must always pass sequence name (can't be nil)
            -- Set rate to 0 only when sequence changes to avoid resetting playback rate
            if clone.currentSeq ~= frame.seqName then
                clone.holo:setAnimation(frame.seqName, frame.animProgress, 0)
                clone.currentSeq = frame.seqName
            else
                clone.holo:setAnimation(frame.seqName, frame.animProgress)
            end

            clone.currentFrame = targetFrame
        end
    end
end)

-- Key input
hook.add("inputPressed", "clone_controls", function(key)
    if key == KEY.PAD_1 then
        spawnClone()
    elseif key == KEY.PAD_2 then
        removeLastClone()
    end
end)

print("[Movement Recorder] Loaded! Recording started.")
print("[Movement Recorder] Numpad 1: Spawn clone | Numpad 2: Remove last clone")
