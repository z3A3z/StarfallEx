--@name Movement Recorder & Cloner
--@author
--@shared

--[[
    Press NUM 1: Spawn a hologram clone that replays your recorded movements
    Press NUM 2: Remove the last spawned clone

    The code continuously records your position, angles, model, and animation.
    All clones loop the same recording path indefinitely.
]]

if SERVER then
    -- Server: Recording data
    local recording = {}
    local recordingStartTime = timer.curtime()
    local clones = {}
    local nextRecordTime = 0
    local recordInterval = 0.05  -- Record at 20 fps to save CPU

    -- Record owner's movement on server (throttled to save CPU)
    hook.add("think", "record_movement", function()
        local ply = owner()
        if not ply:isValid() then return end

        -- Throttle recording to reduce CPU usage
        local curTime = timer.curtime()
        if curTime < nextRecordTime then return end
        nextRecordTime = curTime + recordInterval

        -- Get body angles (only yaw)
        local bodyAng = ply:getAngles()
        local bodyYaw = Angle(0, bodyAng.y, 0)

        -- Get animation and pose data
        local seqID = ply:getSequence()
        local moveX = ply:getPose("move_x")
        local moveY = ply:getPose("move_y")

        local frame = {
            pos = ply:getPos(),
            ang = bodyYaw,
            seq = seqID,
            moveX = moveX,
            moveY = moveY,
            model = ply:getModel(),
            time = curTime - recordingStartTime
        }

        table.insert(recording, frame)
    end)

    -- Spawn a clone
    net.receive("spawnClone", function(len, ply)
        -- Only allow the chip owner to spawn clones
        if ply ~= owner() then return end

        if #recording == 0 then
            net.start("cloneMessage")
            net.writeString("[Clone] No recording data available!")
            net.send(ply)
            return
        end

        local startFrame = recording[1]

        -- Create hologram
        local holo = holograms.create(
            startFrame.pos,
            startFrame.ang,
            startFrame.model,
            Vector(1, 1, 1)
        )

        if not holo then
            net.start("cloneMessage")
            net.writeString("[Clone] Failed to create hologram!")
            net.send(ply)
            return
        end

        -- Set initial animation and pose
        holo:setAnimation(startFrame.seq, 0, 1)
        holo:setPose("move_x", startFrame.moveX)
        holo:setPose("move_y", startFrame.moveY)

        -- Store clone data
        local cloneData = {
            holo = holo,
            startTime = timer.curtime(),
            currentFrame = 1,
            currentSeq = startFrame.seq
        }

        table.insert(clones, cloneData)

        net.start("cloneMessage")
        net.writeString("[Clone] Spawned clone #" .. #clones .. " (using live recording with " .. #recording .. " frames)")
        net.send(ply)
    end)

    -- Remove last clone
    net.receive("removeClone", function(len, ply)
        -- Only allow the chip owner to remove clones
        if ply ~= owner() then return end

        if #clones == 0 then
            net.start("cloneMessage")
            net.writeString("[Clone] No clones to remove!")
            net.send(ply)
            return
        end

        local lastClone = table.remove(clones)
        if lastClone.holo:isValid() then
            lastClone.holo:remove()
        end

        net.start("cloneMessage")
        net.writeString("[Clone] Removed clone. " .. #clones .. " clones remaining.")
        net.send(ply)
    end)

    -- Update all clones
    hook.add("think", "update_clones", function()
        if #recording == 0 then return end

        for i = #clones, 1, -1 do
            local clone = clones[i]

            if not clone.holo:isValid() then
                table.remove(clones, i)
            else
                local elapsed = timer.curtime() - clone.startTime
                local totalFrames = #recording
                local totalDuration = recording[totalFrames].time

                -- Loop the recording infinitely
                if totalDuration > 0 then
                    elapsed = elapsed % totalDuration
                end

                -- Find frame - start from last frame to avoid full linear search
                local targetFrame = clone.currentFrame or 1

                -- Search forward from last frame
                for j = targetFrame, totalFrames do
                    if recording[j].time > elapsed then
                        targetFrame = j - 1
                        break
                    end
                    targetFrame = j
                end

                -- Handle loop wrap-around (when time resets)
                if elapsed < recording[targetFrame].time then
                    targetFrame = 1
                    for j = 1, totalFrames do
                        if recording[j].time > elapsed then
                            targetFrame = j - 1
                            break
                        end
                        targetFrame = j
                    end
                end

                if targetFrame < 1 then targetFrame = 1 end

                -- Apply the frame data
                local frame = recording[targetFrame]
                clone.holo:setPos(frame.pos)
                clone.holo:setAngles(frame.ang)

                -- Set pose parameters for leg movement
                clone.holo:setPose("move_x", frame.moveX)
                clone.holo:setPose("move_y", frame.moveY)

                -- Only change animation when sequence changes
                if clone.currentSeq ~= frame.seq then
                    clone.holo:setAnimation(frame.seq, 0, 1)
                    clone.currentSeq = frame.seq
                end

                clone.currentFrame = targetFrame
            end
        end
    end)
end

if CLIENT then
    -- Client: Input detection
    hook.add("inputPressed", "clone_controls", function(key)
        if key == KEY.PAD_1 then
            net.start("spawnClone")
            net.send()
        elseif key == KEY.PAD_2 then
            net.start("removeClone")
            net.send()
        end
    end)

    -- Client: Receive messages from server
    net.receive("cloneMessage", function()
        local msg = net.readString()
        print(msg)
    end)

    print("[Movement Recorder] Loaded! Recording started.")
    print("[Movement Recorder] Numpad 1: Spawn clone | Numpad 2: Remove last clone")
end
