--@name Movement Recorder & Cloner
--@author
--@shared

--[[
    Press NUM 1: Spawn a hologram clone that replays your recorded movements
    Press NUM 2: Remove the last spawned clone
    Press NUM 6: Set recording target to owner (yourself)
    Press NUM 7: Start/Resume recording
    Press NUM 8: Pause recording
    Press NUM 9: Set recording target to player you're looking at

    The code continuously records your position, angles, model, and animation.
    All clones loop the same recording path indefinitely.
]]

if SERVER then
    -- Server: Recording data
    local recording = {}
    local recordingStartTime = timer.curtime()
    local clones = {}
    local nextRecordTime = 0
    local recordInterval = 0.05  -- Record at 20 fps to save CPU
    local isRecording = false  -- Recording paused by default - press NUM 7 to start
    local playerColor = nil  -- Store player color once
    local recordTarget = owner()  -- Target to record (default: owner)

    -- Record target's movement on server (throttled to save CPU)
    hook.add("think", "record_movement", function()
        if not isRecording then return end  -- Skip if paused

        local ply = recordTarget
        if not ply or not ply:isValid() then return end

        -- Get player color once on first frame
        if playerColor == nil then
            playerColor = ply:getPlayerColor()
        end

        -- Throttle recording to reduce CPU usage
        local curTime = timer.curtime()
        if curTime < nextRecordTime then return end
        nextRecordTime = curTime + recordInterval

        -- Get body angles (only yaw)
        local bodyAng = ply:getAngles()
        local bodyYaw = Angle(0, bodyAng.y, 0)

        -- Get animation and pose data
        local seqID = ply:getSequence()
        local moveX = ply:getPose("move_x")
        local moveY = ply:getPose("move_y")

        local frame = {
            pos = ply:getPos(),
            ang = bodyYaw,
            seq = seqID,
            moveX = moveX,
            moveY = moveY,
            model = ply:getModel(),
            time = curTime - recordingStartTime
        }

        table.insert(recording, frame)
    end)

    -- Spawn a clone
    net.receive("spawnClone", function(len, ply)
        -- Only allow the chip owner to spawn clones
        if ply ~= owner() then return end

        if #recording == 0 then
            net.start("cloneMessage")
            net.writeString("[Clone] No recording data available!")
            net.send(ply)
            return
        end

        local startFrame = recording[1]

        -- Create hologram
        local holo = holograms.create(
            startFrame.pos,
            startFrame.ang,
            startFrame.model,
            Vector(1, 1, 1)
        )

        if not holo then
            net.start("cloneMessage")
            net.writeString("[Clone] Failed to create hologram!")
            net.send(ply)
            return
        end

        -- Set initial animation and pose
        holo:setAnimation(startFrame.seq, 0, 1)
        holo:setPose("move_x", startFrame.moveX)
        holo:setPose("move_y", startFrame.moveY)

        -- Set player color (stored once at initialization)
        if playerColor then
            holo:setPlayerColor(playerColor)
        end

        -- Store clone data
        local cloneData = {
            holo = holo,
            startTime = timer.curtime(),
            currentFrame = 1,
            currentSeq = startFrame.seq
        }

        table.insert(clones, cloneData)

        net.start("cloneMessage")
        net.writeString("[Clone] Spawned clone #" .. #clones .. " (using live recording with " .. #recording .. " frames)")
        net.send(ply)
    end)

    -- Remove last clone
    net.receive("removeClone", function(len, ply)
        -- Only allow the chip owner to remove clones
        if ply ~= owner() then return end

        if #clones == 0 then
            net.start("cloneMessage")
            net.writeString("[Clone] No clones to remove!")
            net.send(ply)
            return
        end

        local lastClone = table.remove(clones)
        if lastClone.holo:isValid() then
            lastClone.holo:remove()
        end

        net.start("cloneMessage")
        net.writeString("[Clone] Removed clone. " .. #clones .. " clones remaining.")
        net.send(ply)
    end)

    -- Resume recording
    net.receive("resumeRecording", function(len, ply)
        if ply ~= owner() then return end

        isRecording = true
        -- Reset start time if this is the first time starting
        if #recording == 0 then
            recordingStartTime = timer.curtime()
        end
        net.start("cloneMessage")
        net.writeString("[Recording] Recording started")
        net.send(ply)
    end)

    -- Pause recording
    net.receive("pauseRecording", function(len, ply)
        if ply ~= owner() then return end

        isRecording = false
        net.start("cloneMessage")
        net.writeString("[Recording] Recording paused")
        net.send(ply)
    end)

    -- Set target to owner
    net.receive("setTargetToOwner", function(len, ply)
        if ply ~= owner() then return end

        recordTarget = owner()
        playerColor = nil  -- Reset player color to capture new target's color
        net.start("cloneMessage")
        net.writeString("[Target] Recording owner")
        net.send(ply)
    end)

    -- Set target to player looking at
    net.receive("setTargetToLookAt", function(len, ply)
        if ply ~= owner() then return end

        local trace = ply:getEyeTrace()
        local target = trace.Entity

        if target and target:isValid() and target:isPlayer() then
            recordTarget = target
            playerColor = nil  -- Reset player color to capture new target's color
            net.start("cloneMessage")
            net.writeString("[Target] Recording " .. target:getName())
            net.send(ply)
        else
            net.start("cloneMessage")
            net.writeString("[Target] No player found! Look at a player and press NUM 9")
            net.send(ply)
        end
    end)

    -- Update all clones
    hook.add("think", "update_clones", function()
        if #recording == 0 then return end

        for i = #clones, 1, -1 do
            local clone = clones[i]

            if not clone.holo:isValid() then
                table.remove(clones, i)
            else
                local elapsed = timer.curtime() - clone.startTime
                local totalFrames = #recording
                local totalDuration = recording[totalFrames].time

                -- Loop the recording infinitely
                if totalDuration > 0 then
                    elapsed = elapsed % totalDuration
                end

                -- Find frame - start from last frame to avoid full linear search
                local targetFrame = clone.currentFrame or 1

                -- Search forward from last frame
                for j = targetFrame, totalFrames do
                    if recording[j].time > elapsed then
                        targetFrame = j - 1
                        break
                    end
                    targetFrame = j
                end

                -- Handle loop wrap-around (when time resets)
                if elapsed < recording[targetFrame].time then
                    targetFrame = 1
                    for j = 1, totalFrames do
                        if recording[j].time > elapsed then
                            targetFrame = j - 1
                            break
                        end
                        targetFrame = j
                    end
                end

                if targetFrame < 1 then targetFrame = 1 end

                -- Apply the frame data
                local frame = recording[targetFrame]
                clone.holo:setPos(frame.pos)
                clone.holo:setAngles(frame.ang)

                -- Set pose parameters for leg movement
                clone.holo:setPose("move_x", frame.moveX)
                clone.holo:setPose("move_y", frame.moveY)

                -- Only change animation when sequence changes
                if clone.currentSeq ~= frame.seq then
                    clone.holo:setAnimation(frame.seq, 0, 1)
                    clone.currentSeq = frame.seq
                end

                clone.currentFrame = targetFrame
            end
        end
    end)
end

if CLIENT then
    -- Client: Track holograms for footstep sounds
    local trackedHolos = {}  -- [hologram] = {lastPos, lastStepTime}
    local maxSoundsPerFrame = 1  -- Strict limit to prevent burst

    -- Monitor holograms and play footstep sounds client-side
    hook.add("think", "clone_footsteps", function()
        -- Check if we have sounds available before processing
        if sound.emitSoundsLeft() < 5 then return end

        local myPos = player():getPos()
        local soundsPlayed = 0

        -- Find all nearby holograms
        for _, ent in ipairs(find.all()) do
            if ent:getClass() == "starfall_hologram" then
                local dist = ent:getPos():getDistance(myPos)

                -- Only process holograms within hearing range
                if dist < 1000 then
                    local pos = ent:getPos()

                    if not trackedHolos[ent] then
                        trackedHolos[ent] = {
                            lastPos = pos,
                            lastStepTime = 0
                        }
                    end

                    local tracked = trackedHolos[ent]
                    if not tracked or not tracked.lastPos then
                        tracked = {lastPos = pos, lastStepTime = 0}
                        trackedHolos[ent] = tracked
                    end

                    local movement = pos - tracked.lastPos

                    -- Check if hologram moved (horizontal movement)
                    local speed = Vector(movement.x, movement.y, 0):getLength()

                    if speed > 2 then  -- Moving threshold
                        local curTime = timer.curtime()
                        local stepInterval = 0.3

                        if curTime - tracked.lastStepTime >= stepInterval and soundsPlayed < maxSoundsPerFrame then
                            -- Play random footstep sound from the hologram entity
                            local footstep = math.random() > 0.5 and "player/footsteps/concrete1.wav" or "player/footsteps/concrete2.wav"
                            ent:emitSound(footstep, 70, 100, 1)
                            tracked.lastStepTime = curTime
                            soundsPlayed = soundsPlayed + 1
                        end
                    end

                    tracked.lastPos = pos
                else
                    -- Remove from tracking if too far
                    trackedHolos[ent] = nil
                end
            end
        end

        -- Clean up invalid holograms
        for ent, _ in pairs(trackedHolos) do
            if not ent:isValid() then
                trackedHolos[ent] = nil
            end
        end
    end)

    -- Client: Input detection
    hook.add("inputPressed", "clone_controls", function(key)
        if key == KEY.PAD_1 then
            net.start("spawnClone")
            net.send()
        elseif key == KEY.PAD_2 then
            net.start("removeClone")
            net.send()
        elseif key == KEY.PAD_6 then
            net.start("setTargetToOwner")
            net.send()
        elseif key == KEY.PAD_7 then
            net.start("resumeRecording")
            net.send()
        elseif key == KEY.PAD_8 then
            net.start("pauseRecording")
            net.send()
        elseif key == KEY.PAD_9 then
            net.start("setTargetToLookAt")
            net.send()
        end
    end)

    -- Client: Receive messages from server
    net.receive("cloneMessage", function()
        local msg = net.readString()
        print(msg)
    end)

    print("[Movement Recorder] Loaded! Recording paused.")
    print("[Controls] 1: Spawn clone | 2: Remove clone")
    print("[Controls] 6: Record owner | 9: Record target")
    print("[Controls] 7: Start/Resume recording | 8: Pause recording")
end
